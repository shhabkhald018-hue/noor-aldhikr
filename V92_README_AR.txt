إذا لم يكن بالمشروع مالك حالي، أنشئ الحساب العادي أولًا وتأكد من ظهوره في Authentication > Users، ثم نفّذ المقطع التالي مرة واحدة فقط من SQL Editor بعد استبدال البريد الواضح ببريد الحساب الصحيح. لا تشغّله لبريد غير متحقق منه:

do $$
declare
  owner_user_id uuid;
  owner_email text := 'ضع-بريد-المالك-هنا@example.com';
begin
  select id into owner_user_id
  from auth.users
  where lower(email) = lower(owner_email)
  limit 1;

  if owner_user_id is null then
    raise exception 'لم يوجد حساب بهذا البريد';
  end if;

  insert into public.profiles(id, email, display_name, role, active)
  values (owner_user_id, owner_email, 'مالك الموقع', 'owner', true)
  on conflict (id) do update
  set email = excluded.email, role = 'owner', active = true, updated_at = now();

  insert into public.user_roles(user_id, role)
  values (owner_user_id, 'owner')
  on conflict (user_id) do update
  set role = 'owner', updated_at = now();
end $$;

هذا الإجراء متعمد داخل SQL Editor فقط؛ لا توجد وظيفة عامة في المتصفح تستطيع منح دور owner.
